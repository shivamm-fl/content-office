def moveZeroes(nums):
w = 0
for r in range(len(nums)):
if nums[r] != 0:
nums[w], nums[r] = nums[r], nums[w]
w += 1
>>> moveZeroes([0, 1, 0, 3, 12])

def removeDuplicates(nums):
w = 1
for r in range(1, len(nums)):
if nums[r] != nums[w - 1]:
nums[w] = nums[r]
w += 1
return w
